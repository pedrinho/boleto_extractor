# Runtime dependencies.
#
# Versions are pinned exactly: unpinned ranges meant no two builds were alike, which
# is how a working deploy could break without a single line of code changing.
# Every package here ships a self-contained wheel - there is no native system
# library to install, which is what allows deployment to hosts without a package
# manager (Vercel) as well as slim containers.

# PDF processing
PyPDF2==3.0.1
pdfplumber==0.11.7
PyMuPDF==1.26.3

# Barcode reading (Interleaved 2 of 5). Replaces pyzbar, which needed the libzbar
# C library installed out-of-band via apt-get/brew.
zxing-cpp==3.1.1

# Imaging
Pillow==11.3.0
numpy==2.2.6

# CLI clipboard support
pyperclip==1.9.0

# Web framework. Vercel serves the WSGI `app` object directly and ignores
# gunicorn; the container image (Render) uses it as its WSGI server.
Flask==3.1.2
Werkzeug==3.1.3
gunicorn==26.0.0
